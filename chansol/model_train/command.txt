# =============================================================================
# official lerobot-train usage
# =============================================================================
#
# note:
# - official lerobot-train itself uses --dataset.root for local paths
# - there is no --dataset_root alias in the original command
# - because official args are policy/environment dependent, a single command cannot
#   literally contain every mutually-exclusive arg at once
# - examples below show the official usage pattern directly

  --dataset.repo_id=local/Isaacsim/OMY_apple_picking/auto_fixed_place_ALL \

  --policy.type=act \
  --policy.device=cuda \
  --output_dir=/nas/AI_Checkpoints/VLA/outputs/train/act_official_cli \
  --job_name=act_official_cli \
  --batch_size=16 \
  --steps=100000 \
  --num_workers=4 \
  --log_freq=20 \
  --save_checkpoint=true \
  --save_freq=10000 \
  --wandb.enable=true \
  --wandb.project=Isaacsim_OMY_apple_picking_auto \
  --wandb.mode=online
